cmake_minimum_required(VERSION 3.24)
project(JumpingFlashCollection LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(COLLECTION_UI_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../robbit-mon-dieu/.workbench/recomp-ui" CACHE PATH "Pinned native UI sources")
set(COLLECTION_FRAMEWORK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../robbit-mon-dieu/.workbench/psxrecomp" CACHE PATH "Pinned framework sources")
find_package(SDL3 CONFIG REQUIRED)
set(IMGUI "${COLLECTION_UI_ROOT}/src/third_party/imgui")
add_executable(JumpingFlashCollection WIN32
    src/main.cpp src/model.cpp
    "${IMGUI}/imgui.cpp" "${IMGUI}/imgui_draw.cpp" "${IMGUI}/imgui_tables.cpp" "${IMGUI}/imgui_widgets.cpp"
    "${IMGUI}/backends/imgui_impl_sdl3.cpp" "${IMGUI}/backends/imgui_impl_opengl3.cpp")
target_include_directories(JumpingFlashCollection PRIVATE src runtime "${IMGUI}" "${IMGUI}/backends"
    "${COLLECTION_UI_ROOT}/src/third_party" "${COLLECTION_FRAMEWORK_ROOT}/recompiler/lib/toml11")
target_compile_definitions(JumpingFlashCollection PRIVATE SDL_MAIN_HANDLED=1)
target_link_libraries(JumpingFlashCollection PRIVATE SDL3::SDL3-static opengl32)
if(MINGW)
    target_link_options(JumpingFlashCollection PRIVATE -static -static-libgcc -static-libstdc++)
endif()
add_executable(collection-model-test tests/model_test.cpp src/model.cpp)
target_include_directories(collection-model-test PRIVATE src runtime "${COLLECTION_FRAMEWORK_ROOT}/recompiler/lib/toml11")
if(MINGW)
    target_link_options(collection-model-test PRIVATE -static)
endif()
enable_testing()
add_test(NAME collection-model COMMAND collection-model-test "${CMAKE_CURRENT_SOURCE_DIR}/catalog.toml")

target_include_directories(collection-robbit-controls-test PRIVATE runtime)

add_test(NAME collection-robbit-control-guards COMMAND collection-robbit-controls-test)
